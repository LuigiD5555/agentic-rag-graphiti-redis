48def _vector_store_settings(config: Any, alias: str) -> Mapping[str, Any]: 

49 """ 

50 Resolve the vector store settings for a given alias from the settings object. 

51 

52 Args: 

53 config: Application settings object or module. Must expose VECTOR_STORES (dict-like). 

54 alias: The vector store alias (e.g., "default"). 

55 

56 Returns: 

57 A dict-like mapping with the store configuration. 

58 

59 Raises: 

60 TypeError: If VECTOR_STORES or the alias config are not dicts. 

61 KeyError: If the alias does not exist. 

62 """ 

63 stores = getattr(config, "VECTOR_STORES", None) or {} 

64 if not isinstance(stores, dict): 

65 raise TypeError("VECTOR_STORES must be a dict mapping aliases to dict settings") 

66 if alias not in stores: 

67 available = ", ".join(sorted(stores.keys())) or "<none>" 

68 raise KeyError(f"Unknown VECTOR_STORES alias '{alias}'. Available: {available}") 

69 store_cfg = stores[alias] 

70 if not isinstance(store_cfg, dict): 

71 raise TypeError(f"VECTOR_STORES['{alias}'] must be a dict of settings") 

72 return store_cfg

75def _config_with_overrides( 

76 config: Any, 

77 overrides: Mapping[str, Any], 

78 key_map: Mapping[str, str], 

79 allowed_passthrough: Iterable[str] | None = None, 

80) -> Any: 

81 """ 

82 Create a new settings-like object with selected overrides applied. 

83 

84 Supports: 

85 - Pydantic settings objects that provide `model_copy(update=...)` 

86 - Python modules (e.g., `import src.settings as settings`) 

87 - Plain objects with attributes 

88 

89 Args: 

90 config: Settings object or module. 

91 overrides: Raw store config dict (Django-style). 

92 key_map: Mapping from Django-style keys to settings attribute names. 

93 allowed_passthrough: Keys that are allowed in overrides but should be ignored here. 

94 

95 Returns: 

96 A settings-like object with updated attributes. 

97 

98 Raises: 

99 ValueError: If an unsupported override key is provided. 

100 """ 

101 update: dict[str, Any] = {} 

102 allowed = set(allowed_passthrough or ()) 

103 

104 for raw_key, raw_value in overrides.items(): 

105 key_upper = str(raw_key or "").upper() 

106 if key_upper in {"BACKEND", "ENGINE"}: 

107 continue 

108 if key_upper in allowed: 

109 continue 

110 if key_upper == "OPTIONS": 

111 # OPTIONS is handled elsewhere (normalized before reaching here) 

112 continue 

113 if key_upper not in key_map: 

114 continue 

115 update[key_map[key_upper]] = raw_value 

116 

117 if not update: 

118 return config 

119 

120 # Case 1: Pydantic v2 style settings 

121 model_copy = getattr(config, "model_copy", None) 

122 if callable(model_copy): 

123 return model_copy(update=update) 

124 

125 # Case 2: Module or plain object -> build a namespace copy 

126 # We copy public attributes to avoid mutating the original module/object. 

127 base_attributes: dict[str, Any] = {} 

128 try: 

129 # For modules, vars(config) works; for many objects too. 

130 base_attributes.update(vars(config)) 

131 except TypeError: 

132 # Fallback for objects without __dict__ 

133 for name in dir(config): 

134 if name.startswith("_"): 

135 continue 

136 try: 

137 base_attributes[name] = getattr(config, name) 

138 except Exception: 

139 continue 

140 

141 base_attributes.update(update) 

142 return types.SimpleNamespace(**base_attributes)

145def _normalize_vector_store_cfg(store_cfg: Mapping[str, Any], config: Any) -> Mapping[str, Any]: 

146 """ 

147 Accept Django-like keys (ENGINE/HOST/PORT/OPTIONS) and synthesize URL when needed. 

148 

149 Args: 

150 store_cfg: Raw store configuration mapping. 

151 config: Settings object/module, used as fallback for WEAVIATE_URL. 

152 

153 Returns: 

154 A normalized store configuration mapping. 

155 """ 

156 normalized = dict(store_cfg) 

157 options = normalized.get("OPTIONS") or {} 

158 if isinstance(options, dict): 

159 if options.get("GRPC_PORT") is not None: 

160 normalized.setdefault("GRPC_PORT", options["GRPC_PORT"]) 

161 if options.get("CONNECT_RETRIES") is not None: 

162 normalized.setdefault("CONNECT_RETRIES", options["CONNECT_RETRIES"]) 

163 if options.get("CONNECT_BACKOFF") is not None: 

164 normalized.setdefault("CONNECT_BACKOFF", options["CONNECT_BACKOFF"]) 

165 if options.get("MULTI_TENANCY") is not None: 

166 normalized.setdefault("MULTI_TENANCY", options["MULTI_TENANCY"]) 

167 if options.get("DEFAULT_TENANT") is not None: 

168 normalized.setdefault("DEFAULT_TENANT", options["DEFAULT_TENANT"]) 

169 if options.get("TIMEOUT") is not None: 

170 normalized.setdefault("TIMEOUT", options["TIMEOUT"]) 

171 

172 if "URL" not in normalized: 

173 host = normalized.get("HOST") 

174 port = normalized.get("PORT") 

175 scheme = normalized.get("SCHEME") or "http" 

176 path = normalized.get("PATH") or "" 

177 if host or port: 

178 host = host or "localhost" 

179 port = port or 8080 

180 url = f"{scheme}://{host}:{port}" 

181 if path: 

182 url = f"{url}/{str(path).lstrip('/')}" 

183 normalized["URL"] = url 

184 

185 if "NAME" in normalized and "CLASS" not in normalized: 

186 normalized["CLASS"] = normalized["NAME"] 

187 

188 return normalized

191def _build_weaviate_repository(config: Any, store_cfg: Mapping[str, Any], alias: str) -> VectorInterface: 

192 """ 

193 Build the default Weaviate repository backend. 

194 

195 This is the only backend implemented for now, but the registry allows 

196 more in the future. 

197 """ 

198 cfg = _config_with_overrides( 

199 config, 

200 store_cfg, 

201 { 

202 "URL": "WEAVIATE_URL", 

203 "API_KEY": "WEAVIATE_API_KEY", 

204 "CLASS": "WEAVIATE_CLASS", 

205 "MULTI_TENANCY": "WEAVIATE_MULTI_TENANCY", 

206 "DEFAULT_TENANT": "WEAVIATE_DEFAULT_TENANT", 

207 "TIMEOUT": "WEAVIATE_TIMEOUT", 

208 "GRPC_PORT": "WEAVIATE_GRPC_PORT", 

209 "CONNECT_RETRIES": "WEAVIATE_CONNECT_RETRIES", 

210 "CONNECT_BACKOFF": "WEAVIATE_CONNECT_BACKOFF", 

211 }, 

212 allowed_passthrough={ 

213 "OPTIONS", 

214 "HOST", 

215 "PORT", 

216 "SCHEME", 

217 "PATH", 

218 "NAME", 

219 }, 

220 ) 

221 

222 weaviate_url = getattr(cfg, "WEAVIATE_URL", None) or store_cfg.get("URL") or "http://localhost:8080" 

223 weaviate_api_key = getattr(cfg, "WEAVIATE_API_KEY", None) or store_cfg.get("API_KEY") 

224 weaviate_class = getattr(cfg, "WEAVIATE_CLASS", None) or store_cfg.get("CLASS") or "RagDocument" 

225 weaviate_timeout = getattr(cfg, "WEAVIATE_TIMEOUT", None) or store_cfg.get("TIMEOUT") or 120 

226 weaviate_multi_tenancy = bool( 

227 getattr(cfg, "WEAVIATE_MULTI_TENANCY", False) or store_cfg.get("MULTI_TENANCY") or False 

228 ) 

229 weaviate_default_tenant = getattr(cfg, "WEAVIATE_DEFAULT_TENANT", None) or store_cfg.get("DEFAULT_TENANT") or "default" 

230 weaviate_grpc_port = getattr(cfg, "WEAVIATE_GRPC_PORT", None) or store_cfg.get("GRPC_PORT") 

231 skip_init_checks = bool(getattr(cfg, "WEAVIATE_SKIP_INIT_CHECKS", False)) 

232 

233 from weaviate.classes.init import AdditionalConfig, Timeout 

234 from src.backends.storage.vector.weaviate_repository.repository import WeaviateRepository 

235 from src.backends.storage.vector.weaviate_repository.schema import SchemaManager 

236 

237 additional = AdditionalConfig(timeout=Timeout(init=weaviate_timeout, query=weaviate_timeout)) 

238 

239 parsed_url = urlparse(weaviate_url) 

240 host = parsed_url.hostname or "localhost" 

241 port = parsed_url.port or (443 if parsed_url.scheme == "https" else 8080) 

242 use_https = parsed_url.scheme == "https" 

243 grpc_port = weaviate_grpc_port or (50051 if not use_https else 443) 

244 

245 import weaviate 

246 

247 if weaviate_api_key: 

248 client = weaviate.connect_to_custom( 

249 http_host=host, 

250 http_port=port, 

251 http_secure=use_https, 

252 grpc_host=host, 

253 grpc_port=grpc_port, 

254 grpc_secure=use_https, 

255 auth_credentials=weaviate.auth.AuthApiKey(api_key=weaviate_api_key), 

256 additional_config=additional, 

257 skip_init_checks=skip_init_checks, 

258 ) 

259 else: 

260 client = weaviate.connect_to_custom( 

261 http_host=host, 

262 http_port=port, 

263 http_secure=use_https, 

264 grpc_host=host, 

265 grpc_port=grpc_port, 

266 grpc_secure=use_https, 

267 additional_config=additional, 

268 skip_init_checks=skip_init_checks, 

269 ) 

270 

271 schema_manager = SchemaManager( 

272 client=client, 

273 cfg=cfg, 

274 class_name=weaviate_class, 

275 multitenant_enabled=weaviate_multi_tenancy, 

276 default_tenant=weaviate_default_tenant if weaviate_multi_tenancy else None, 

277 class_properties_provider=_get_rag_document_properties, 

278 ) 

279 schema_manager.ensure_class() 

280 

281 return WeaviateRepository(schema=schema_manager)

288def get_vector_store(config: Any, alias: str = "default") -> VectorInterface: 

289 """ 

290 Create the vector store backend selected in settings. 

291 

292 Django-like usage: 

293 VECTOR_STORES = { 

294 "default": {"ENGINE": "weaviate"}, 

295 "analytics": {"ENGINE": "weaviate", "URL": "..."}, 

296 } 

297 

298 Args: 

299 config: Application settings object or module. 

300 alias: Vector store alias to use. 

301 

302 Returns: 

303 An instance implementing VectorInterface. 

304 """ 

305 store_cfg = _normalize_vector_store_cfg(_vector_store_settings(config, alias), config) 

306 backend = (store_cfg.get("BACKEND") or store_cfg.get("ENGINE") or "").strip().lower() 

307 if not backend: 

308 backend = (getattr(config, "VECTOR_BACKEND", None) or "weaviate").strip().lower() 

309 

310 factory = get_vector_store_factory(backend) 

311 return factory(config, store_cfg, alias)
