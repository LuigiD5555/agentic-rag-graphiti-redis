126 @staticmethod 

127 def detect_language(text: str) -> str: 

128 """Detect the language of the given text. 

129 

130 Args: 

131 text: Input text to detect language for. 

132 

133 Returns: 

134 ISO 639-1 language code (e.g., 'en', 'es', 'zh-cn', 'ja', 'ko', 'ru'). 

135 """ 

136 if not text or not text.strip(): 

137 return 'en' # Default to English 

138 

139 try: 

140 lang_code = detect(text) 

141 log.debug("Detected language: %s for text: %s", lang_code, text[:50]) 

142 return lang_code 

143 except LangDetectException as e: 

144 log.warning("Failed to detect language: %s. Defaulting to English.", e) 

145 return 'en'

147 @classmethod 

148 def get_system_prompt(cls, language_code: str) -> str: 

149 """Get system prompt for the detected language. 

150 

151 Args: 

152 language_code: ISO 639-1 language code. 

153 

154 Returns: 

155 System prompt in the appropriate language. 

156 """ 

157 # Try exact match first 

158 if language_code in cls.SYSTEM_PROMPTS: 

159 return cls.SYSTEM_PROMPTS[language_code] 

160 

161 # Handle Chinese variants 

162 if language_code.startswith('zh'): 

163 return cls.SYSTEM_PROMPTS.get('zh-cn', cls.SYSTEM_PROMPTS['en']) 

164 

165 # Default to English 

166 log.debug("No system prompt for language %s, using English", language_code) 

167 return cls.SYSTEM_PROMPTS['en']

169 @classmethod 

170 def get_language_name(cls, language_code: str) -> str: 

171 """Get full language name from code. 

172 

173 Args: 

174 language_code: ISO 639-1 language code. 

175 

176 Returns: 

177 Full language name. 

178 """ 

179 return cls.LANGUAGE_NAMES.get(language_code, f"Unknown ({language_code})")

182 def add_language_instruction(cls, system_prompt: str, language_code: str) -> str: 

183 """Add explicit instruction to respond in the detected language.

185 Args:

187 language_code: Detected language code. 

188

190 Enhanced system prompt with language instruction. 

191 """ 

192 if language_code == 'en': 

193 return system_prompt # No need for explicit instruction in English
