146 def classify(self, query: str) -> IntentType: 

147 """Classify query intent for routing decisions. 

148 

149 Args: 

150 query: User query string. 

151 

152 Returns: 

153 Intent type: SMALL_TALK, PERSONAL_CHAT, CONTROL, GENERAL_KNOWLEDGE, or PERSONAL_KB 

154 """ 

155 query_lower = query.lower().strip() 

156 query_len = len(query) 

157 

158 # Detect language hint 

159 lang_hint = self.lang_detector.detect_language_hint(query) or 'en' 

160 

161 log.debug("Classifying intent: query='%s' (len=%d, lang_hint=%s)", 

162 query[:50], query_len, lang_hint) 

163 

164 # Step 1: Check for PERSONAL_KB patterns (highest priority) 

165 # These indicate explicit reference to user's documents 

166 if self._match_patterns(query_lower, self.PERSONAL_KB_PATTERNS, lang_hint): 

167 log.info("Intent: PERSONAL_KB (references user documents)") 

168 return "PERSONAL_KB" 

169 

170 # Step 2: Check for SMALL_TALK (greetings, thanks, simple affirmations) 

171 # Very short queries (<= 20 chars) with known patterns 

172 if query_len <= 20: 

173 if self._match_patterns(query_lower, self.GREETINGS, lang_hint): 

174 log.info("Intent: SMALL_TALK (greeting)") 

175 return "SMALL_TALK" 

176 if self._match_patterns(query_lower, self.THANKS, lang_hint): 

177 log.info("Intent: SMALL_TALK (thanks)") 

178 return "SMALL_TALK" 

179 if self._match_patterns(query_lower, self.GOODBYES, lang_hint): 

180 log.info("Intent: SMALL_TALK (goodbye)") 

181 return "SMALL_TALK" 

182 

183 # Step 3: Check for affirmations/negations (conversational noise) 

184 if query_len <= 30: 

185 if self._match_patterns(query_lower, self.AFFIRMATIONS, lang_hint): 

186 log.info("Intent: SMALL_TALK (affirmation)") 

187 return "SMALL_TALK" 

188 if self._match_patterns(query_lower, self.NEGATIONS, lang_hint): 

189 log.info("Intent: SMALL_TALK (negation)") 

190 return "SMALL_TALK" 

191 

192 # Step 4: Check for PERSONAL_CHAT (bot identity/feelings) 

193 # Questions about the bot itself, not about documents 

194 if query_len <= 50: 

195 if self._match_patterns(query_lower, self.BOT_NAME_PATTERNS, lang_hint): 

196 log.info("Intent: PERSONAL_CHAT (bot name/identity)") 

197 return "PERSONAL_CHAT" 

198 if self._match_patterns(query_lower, self.BOT_FEELINGS_PATTERNS, lang_hint): 

199 log.info("Intent: PERSONAL_CHAT (bot feelings/opinions)") 

200 return "PERSONAL_CHAT" 

201 

202 # Step 5: Check for CONTROL commands 

203 if self._match_patterns(query_lower, self.CONTROL_PATTERNS, lang_hint): 

204 log.info("Intent: CONTROL (format/style command)") 

205 return "CONTROL" 

206 

207 # Step 6: Default to GENERAL_KNOWLEDGE 

208 # Informational queries that don't reference user's documents 

209 log.info("Intent: GENERAL_KNOWLEDGE (default for informational queries)") 

210 return "GENERAL_KNOWLEDGE"

212 def _match_patterns(self, query_lower: str, pattern_dict: dict, lang_hint: str) -> bool: 

213 """Check if query matches any pattern for the detected language. 

214 

215 Args: 

216 query_lower: Lowercased query. 

217 pattern_dict: Dict of {lang: [patterns]}. 

218 lang_hint: Detected language hint. 

219 

220 Returns: 

221 True if any pattern matches. 

222 """ 

223 # Check patterns for detected language 

224 if lang_hint in pattern_dict: 

225 for pattern in pattern_dict[lang_hint]: 

226 if pattern in query_lower: 

227 return True 

228 

229 # Fallback: check all languages if no match in detected language 

230 # (handles multilingual queries and misdetection) 

231 for patterns in pattern_dict.values(): 

232 for pattern in patterns: 

233 if pattern in query_lower: 

234 return True 

235 

236 return False

238 def should_use_rag(self, intent: IntentType) -> bool: 

239 """Determine if RAG retrieval should be used for this intent. 

240 

241 Args: 

242 intent: Classified intent type. 

243 

244 Returns: 

245 True if RAG should be used, False otherwise. 

246 """ 

247 # Use RAG for PERSONAL_KB (explicit references) and GENERAL_KNOWLEDGE (try RAG first) 

248 # Skip RAG only for SMALL_TALK, PERSONAL_CHAT, and CONTROL 

249 return intent in ("PERSONAL_KB", "GENERAL_KNOWLEDGE")
